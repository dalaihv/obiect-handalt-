SELECT 
    s.wcode AS "Ажилтны код",
    w.wlast AS "Овог",
    w.wname AS "Нэр",
    s.salary AS "Үндсэн цалин",
    s.Tax AS "Хаоат",
    s.ndsh AS "НДШ",
    s.other AS "Бусад суутгал",
    su.suutname AS "Суутгалын нэр",
    s.payment AS "Урьдчилгаа",
    s.garolgo AS "Гарт олгох дүн"
FROM salary s
JOIN worker w ON s.wcode = w.wcode
LEFT JOIN nemegdel n ON s.nemcode = n.nemcode
LEFT JOIN suutgal su ON s.suutcode = su.suutcode;
